```csharp
    }

/// <summary>
/// Branch prediction hint annotation
/// Added by ProfileGuidedBranchPredictionPass
/// </summary>
public class IrBranchHint
{
    public BranchPath LikelyPath { get; set; }
    public double Probability { get; set; }
    public PredictionStrength Strength { get; set; }

    public IrBranchHint(BranchPath likelyPath, double probability)
    {
        LikelyPath = likelyPath;
        Probability = probability;
        Strength = CalculateStrength(probability);
    }

    private static PredictionStrength CalculateStrength(double prob)
    {
        if (prob > 0.95 || prob < 0.05)
            return PredictionStrength.Strong;
        if (prob > 0.85 || prob < 0.15)
            return PredictionStrength.Moderate;
        return PredictionStrength.Weak;
    }
}

public enum BranchPath
{
    True,
    False
}

public enum PredictionStrength
{
    Weak,     // 70-85% prediction accuracy
    Moderate, // 85-95% prediction accuracy
    Strong    // >95% prediction accuracy
}
```

## 2. Profile Data Structures

### 2.1 Complete ProfileData Implementation

```csharp
namespace Novus.Optimizer;

using System;
using System.IO;
using System.Text;
using System.Security.Cryptography;

public class ProfileData
{
    public const uint MAGIC = 0x4E50474F; // "NPGO"
    public const ushort VERSION = 0x0001;

    public byte[] SourceHash { get; set; } = new byte[32];
    public DateTime Timestamp { get; set; }
    public string CompilerVersion { get; set; } = "";

    public Dictionary<string, FunctionProfile> Functions { get; } = new();
    public Dictionary<int, BranchProfile> Branches { get; } = new();
    public Dictionary<int, LoopProfile> Loops { get; } = new();
    public Dictionary<string, CallSiteProfile> CallSites { get; } = new();

    /// <summary>
    /// Load profile data from .pgo file
    /// </summary>
    public static ProfileData Load(string path)
    {
        if (!File.Exists(path))
            throw new FileNotFoundException($"Profile data file not found: {path}");

        using var stream = File.OpenRead(path);
        using var reader = new BinaryReader(stream);

        // Read and verify header
        uint magic = reader.ReadUInt32();
        if (magic != MAGIC)
            throw new InvalidDataException($"Invalid .pgo file: expected magic 0x{MAGIC:X8}, got 0x{magic:X8}");

        ushort version = reader.ReadUInt16();
        if (version != VERSION)
            throw new InvalidDataException($"Unsupported .pgo version: {version}, expected {VERSION}");

        var profile = new ProfileData
        {
            SourceHash = reader.ReadBytes(32),
            Timestamp = DateTimeOffset.FromUnixTimeSeconds(reader.ReadInt64()).DateTime
        };

        byte compilerVersionLen = reader.ReadByte();
        profile.CompilerVersion = Encoding.UTF8.GetString(reader.ReadBytes(compilerVersionLen));

        // Read function profiles
        ushort numFunctions = reader.ReadUInt16();
        for (int i = 0; i < numFunctions; i++)
        {
            ushort id = reader.ReadUInt16();
            byte nameLen = reader.ReadByte();
            string name = Encoding.UTF8.GetString(reader.ReadBytes(nameLen));
            uint count = reader.ReadUInt32();
            uint cumulativeCycles = reader.ReadUInt32(); // Optional: cycle counter

            profile.Functions[name] = new FunctionProfile
            {
                Id = id,
                Name = name,
                ExecutionCount = count,
                CumulativeCycles = cumulativeCycles
            };
        }

        // Read branch profiles
        ushort numBranches = reader.ReadUInt16();
        for (int i = 0; i < numBranches; i++)
        {
            ushort id = reader.ReadUInt16();
            uint fileId = reader.ReadUInt32();
            uint line = reader.ReadUInt32();
            uint trueCount = reader.ReadUInt32();
            uint falseCount = reader.ReadUInt32();

            profile.Branches[id] = new BranchProfile
            {
                Id = id,
                FileId = fileId,
                Line = line,
                TrueCount = trueCount,
                FalseCount = falseCount
            };
        }

        // Read loop profiles
        ushort numLoops = reader.ReadUInt16();
        for (int i = 0; i < numLoops; i++)
        {
            ushort id = reader.ReadUInt16();
            uint fileId = reader.ReadUInt32();
            uint line = reader.ReadUInt32();
            uint totalIters = reader.ReadUInt32();
            uint execCount = reader.ReadUInt32();
            uint minIters = reader.ReadUInt32();
            uint maxIters = reader.ReadUInt32();

            profile.Loops[id] = new LoopProfile
            {
                Id = id,
                FileId = fileId,
                Line = line,
                TotalIterations = totalIters,
                ExecutionCount = execCount,
                MinIterations = minIters,
                MaxIterations = maxIters
            };
        }

        // Read call site profiles
        ushort numCallSites = reader.ReadUInt16();
        for (int i = 0; i < numCallSites; i++)
        {
            ushort id = reader.ReadUInt16();
            byte callerLen = reader.ReadByte();
            string caller = Encoding.UTF8.GetString(reader.ReadBytes(callerLen));
            byte calleeLen = reader.ReadByte();
            string callee = Encoding.UTF8.GetString(reader.ReadBytes(calleeLen));
            uint count = reader.ReadUInt32();

            string key = $"{caller}->{callee}";
            profile.CallSites[key] = new CallSiteProfile
            {
                Id = id,
                Caller = caller,
                Callee = callee,
                CallCount = count
            };
        }

        return profile;
    }

    /// <summary>
    /// Save profile data to .pgo file
    /// </summary>
    public void Save(string path)
    {
        using var stream = File.Create(path);
        using var writer = new BinaryWriter(stream);

        // Write header
        writer.Write(MAGIC);
        writer.Write(VERSION);
        writer.Write(SourceHash);
        writer.Write((long)new DateTimeOffset(Timestamp).ToUnixTimeSeconds());

        byte[] compilerVersionBytes = Encoding.UTF8.GetBytes(CompilerVersion);
        writer.Write((byte)compilerVersionBytes.Length);
        writer.Write(compilerVersionBytes);

        // Write functions
        writer.Write((ushort)Functions.Count);
        foreach (var func in Functions.Values.OrderBy(f => f.Id))
        {
            writer.Write(func.Id);
            byte[] nameBytes = Encoding.UTF8.GetBytes(func.Name);
            writer.Write((byte)nameBytes.Length);
            writer.Write(nameBytes);
            writer.Write(func.ExecutionCount);
            writer.Write(func.CumulativeCycles);
        }

        // Write branches
        writer.Write((ushort)Branches.Count);
        foreach (var branch in Branches.Values.OrderBy(b => b.Id))
        {
            writer.Write(branch.Id);
            writer.Write(branch.FileId);
            writer.Write(branch.Line);
            writer.Write(branch.TrueCount);
            writer.Write(branch.FalseCount);
        }

        // Write loops
        writer.Write((ushort)Loops.Count);
        foreach (var loop in Loops.Values.OrderBy(l => l.Id))
        {
            writer.Write(loop.Id);
            writer.Write(loop.FileId);
            writer.Write(loop.Line);
            writer.Write(loop.TotalIterations);
            writer.Write(loop.ExecutionCount);
            writer.Write(loop.MinIterations);
            writer.Write(loop.MaxIterations);
        }

        // Write call sites
        writer.Write((ushort)CallSites.Count);
        foreach (var callSite in CallSites.Values.OrderBy(cs => cs.Id))
        {
            writer.Write(callSite.Id);
            byte[] callerBytes = Encoding.UTF8.GetBytes(callSite.Caller);
            writer.Write((byte)callerBytes.Length);
            writer.Write(callerBytes);
            byte[] calleeBytes = Encoding.UTF8.GetBytes(callSite.Callee);
            writer.Write((byte)calleeBytes.Length);
            writer.Write(calleeBytes);
            writer.Write(callSite.CallCount);
        }
    }

    /// <summary>
    /// Verify that profile data matches the current source code
    /// </summary>
    public bool VerifySourceHash(IrModule module, List<string> sourceFiles)
    {
        var currentHash = ComputeSourceHash(sourceFiles);
        return currentHash.SequenceEqual(SourceHash);
    }

    /// <summary>
    /// Compute SHA-256 hash of all source files
    /// </summary>
    public static byte[] ComputeSourceHash(List<string> sourceFiles)
    {
        using var sha256 = SHA256.Create();
        using var ms = new MemoryStream();

        // Hash all source files in deterministic order
        foreach (var file in sourceFiles.OrderBy(f => f))
        {
            if (!File.Exists(file)) continue;

            var content = File.ReadAllBytes(file);
            ms.Write(content, 0, content.Length);
        }

        ms.Position = 0;
        return sha256.ComputeHash(ms);
    }

    /// <summary>
    /// Merge multiple profile data files
    /// Used for multi-run profiling scenarios
    /// </summary>
    public static ProfileData Merge(List<ProfileData> profiles)
    {
        if (profiles.Count == 0)
            throw new ArgumentException("No profiles to merge");

        var merged = new ProfileData
        {
            SourceHash = profiles[0].SourceHash,
            Timestamp = DateTime.UtcNow,
            CompilerVersion = profiles[0].CompilerVersion
        };

        // Merge function profiles (sum execution counts)
        foreach (var profile in profiles)
        {
            foreach (var func in profile.Functions.Values)
            {
                if (merged.Functions.TryGetValue(func.Name, out var existing))
                {
                    existing.ExecutionCount += func.ExecutionCount;
                    existing.CumulativeCycles += func.CumulativeCycles;
                }
                else
                {
                    merged.Functions[func.Name] = new FunctionProfile
                    {
                        Id = func.Id,
                        Name = func.Name,
                        ExecutionCount = func.ExecutionCount,
                        CumulativeCycles = func.CumulativeCycles
                    };
                }
            }
        }

        // Merge branch profiles
        foreach (var profile in profiles)
        {
            foreach (var branch in profile.Branches.Values)
            {
                if (merged.Branches.TryGetValue(branch.Id, out var existing))
                {
                    existing.TrueCount += branch.TrueCount;
                    existing.FalseCount += branch.FalseCount;
                }
                else
                {
                    merged.Branches[branch.Id] = new BranchProfile
                    {
                        Id = branch.Id,
                        FileId = branch.FileId,
                        Line = branch.Line,
                        TrueCount = branch.TrueCount,
                        FalseCount = branch.FalseCount
                    };
                }
            }
        }

        // Merge loop profiles
        foreach (var profile in profiles)
        {
            foreach (var loop in profile.Loops.Values)
            {
                if (merged.Loops.TryGetValue(loop.Id, out var existing))
                {
                    existing.TotalIterations += loop.TotalIterations;
                    existing.ExecutionCount += loop.ExecutionCount;
                    existing.MinIterations = Math.Min(existing.MinIterations, loop.MinIterations);
                    existing.MaxIterations = Math.Max(existing.MaxIterations, loop.MaxIterations);
                }
                else
                {
                    merged.Loops[loop.Id] = new LoopProfile
                    {
                        Id = loop.Id,
                        FileId = loop.FileId,
                        Line = loop.Line,
                        TotalIterations = loop.TotalIterations,
                        ExecutionCount = loop.ExecutionCount,
                        MinIterations = loop.MinIterations,
                        MaxIterations = loop.MaxIterations
                    };
                }
            }
        }

        // Merge call site profiles
        foreach (var profile in profiles)
        {
            foreach (var callSite in profile.CallSites.Values)
            {
                string key = $"{callSite.Caller}->{callSite.Callee}";
                if (merged.CallSites.TryGetValue(key, out var existing))
                {
                    existing.CallCount += callSite.CallCount;
                }
                else
                {
                    merged.CallSites[key] = new CallSiteProfile
                    {
                        Id = callSite.Id,
                        Caller = callSite.Caller,
                        Callee = callSite.Callee,
                        CallCount = callSite.CallCount
                    };
                }
            }
        }

        return merged;
    }

    /// <summary>
    /// Get a summary of the profile data
    /// </summary>
    public ProfileSummary GetSummary()
    {
        return new ProfileSummary
        {
            TotalFunctions = Functions.Count,
            TotalBranches = Branches.Count,
            TotalLoops = Loops.Count,
            TotalCallSites = CallSites.Count,
            HotFunctions = Functions.Values.Count(f => f.IsHot()),
            ColdFunctions = Functions.Values.Count(f => f.IsCold()),
            PredictableBranches = Branches.Values.Count(b => b.IsPredictable()),
            UnrollableLoops = Loops.Values.Count(l => l.ShouldUnroll())
        };
    }
}

public class ProfileSummary
{
    public int TotalFunctions { get; set; }
    public int TotalBranches { get; set; }
    public int TotalLoops { get; set; }
    public int TotalCallSites { get; set; }
    public int HotFunctions { get; set; }
    public int ColdFunctions { get; set; }
    public int PredictableBranches { get; set; }
    public int UnrollableLoops { get; set; }

    public override string ToString()
    {
        var sb = new StringBuilder();
        sb.AppendLine("Profile Summary:");
        sb.AppendLine($"  Functions:           {TotalFunctions} ({HotFunctions} hot, {ColdFunctions} cold)");
        sb.AppendLine($"  Branches:            {TotalBranches} ({PredictableBranches} predictable)");
        sb.AppendLine($"  Loops:               {TotalLoops} ({UnrollableLoops} unrollable)");
        sb.AppendLine($"  Call Sites:          {TotalCallSites}");
        return sb.ToString();
    }
}

public class FunctionProfile
{
    public ushort Id { get; set; }
    public string Name { get; set; } = "";
    public uint ExecutionCount { get; set; }
    public uint CumulativeCycles { get; set; } // Optional: total CPU cycles spent

    public bool IsHot(uint threshold = 1000) => ExecutionCount >= threshold;
    public bool IsCold(uint threshold = 10) => ExecutionCount < threshold;

    public double CyclesPerExecution => ExecutionCount > 0
        ? (double)CumulativeCycles / ExecutionCount
        : 0;
}

public class BranchProfile
{
    public ushort Id { get; set; }
    public uint FileId { get; set; }
    public uint Line { get; set; }
    public uint TrueCount { get; set; }
    public uint FalseCount { get; set; }

    public uint TotalCount => TrueCount + FalseCount;

    public double TrueProbability
    {
        get
        {
            if (TotalCount == 0) return 0.5;
            return (double)TrueCount / TotalCount;
        }
    }

    public bool IsPredictable(double threshold = 0.85)
    {
        double prob = TrueProbability;
        return prob >= threshold || prob <= (1.0 - threshold);
    }

    public bool TruePathIsLikely => TrueProbability >= 0.5;
}

public class LoopProfile
{
    public ushort Id { get; set; }
    public uint FileId { get; set; }
    public uint Line { get; set; }
    public uint TotalIterations { get; set; }
    public uint ExecutionCount { get; set; }
    public uint MinIterations { get; set; }
    public uint MaxIterations { get; set; }

    public double AverageIterations
    {
        get
        {
            if (ExecutionCount == 0) return 0;
            return (double)TotalIterations / ExecutionCount;
        }
    }

    public bool IsConstantIterations => MinIterations == MaxIterations;

    public bool ShouldUnroll(double maxAvgIters = 8.0)
    {
        return ExecutionCount > 100 &&
               AverageIterations > 1.5 &&
               AverageIterations <= maxAvgIters;
    }
}

public class CallSiteProfile
{
    public ushort Id { get; set; }
    public string Caller { get; set; } = "";
    public string Callee { get; set; } = "";
    public uint CallCount { get; set; }

    public bool ShouldInline(uint threshold = 1000) => CallCount >= threshold;
}
```

## 3. Instrumentation Implementation

### 3.1 InstrumentationPass

```csharp
namespace Novus.Optimizer.Passes;

using Novus.IR;

public class InstrumentationPass : IOptimizationPass
{
    private int _nextFunctionId = 0;
    private int _nextBranchId = 0;
    private int _nextLoopId = 0;
    private int _nextCallSiteId = 0;

    private readonly Dictionary<string, ushort> _functionIds = new();
    private readonly List<(ushort Id, string Name)> _functionMetadata = new();
    private readonly List<(ushort Id, uint FileId, uint Line)> _branchMetadata = new();
    private readonly List<(ushort Id, uint FileId, uint Line)> _loopMetadata = new();
    private readonly List<(ushort Id, string Caller, string Callee)> _callSiteMetadata = new();

    public string Name => "PGO Instrumentation";

    public bool Run(IrModule module)
    {
        // Add global counter arrays
        AddGlobalCounters(module);

        // Instrument each function
        foreach (var function in module.Functions)
        {
            if (function.IsExtern) continue;

            // Assign function ID
            ushort funcId = (ushort)_nextFunctionId++;
            _functionIds[function.Name] = funcId;
            _functionMetadata.Add((funcId, function.Name));

            // Instrument function entry
            InstrumentFunctionEntry(function, funcId);

            // Instrument branches
            InstrumentBranches(function);

            // Instrument loops
            InstrumentLoops(function);

            // Instrument call sites
            InstrumentCallSites(function);
        }

        // Add profile dump on exit
        AddProfileDumpAtExit(module);

        // Save metadata for .pgo file generation
        SaveInstrumentationMetadata(module);

        return true;
    }

    private void AddGlobalCounters(IrModule module)
    {
        // Function counters
        var funcCounterType = new IrArrayType(IrIntType.U32, 256);
        module.StaticVariables.Add(new IrStaticVariable(
            "__pgo_func_counter",
            funcCounterType,
            Visibility.Private,
            isMutable: true,
            initialValue: new IrArrayLiteral(funcCounterType)
        ));

        // Branch counters
        var branchCounterType = new IrArrayType(IrIntType.U32, 1024);
        module.StaticVariables.Add(new IrStaticVariable(
            "__pgo_branch_counter",
            branchCounterType,
            Visibility.Private,
            isMutable: true,
            initialValue: new IrArrayLiteral(branchCounterType)
        ));

        // Loop counters (total iterations)
        var loopTotalType = new IrArrayType(IrIntType.U32, 128);
        module.StaticVariables.Add(new IrStaticVariable(
            "__pgo_loop_total",
            loopTotalType,
            Visibility.Private,
            isMutable: true,
            initialValue: new IrArrayLiteral(loopTotalType)
        ));

        // Loop execution counters
        var loopExecType = new IrArrayType(IrIntType.U32, 128);
        module.StaticVariables.Add(new IrStaticVariable(
            "__pgo_loop_executions",
            loopExecType,
            Visibility.Private,
            isMutable: true,
            initialValue: new IrArrayLiteral(loopExecType)
        ));

        // Loop min/max iteration trackers
        var loopMinType = new IrArrayType(IrIntType.U32, 128);
        module.StaticVariables.Add(new IrStaticVariable(
            "__pgo_loop_min",
            loopMinType,
            Visibility.Private,
            isMutable: true,
            initialValue: CreateInitializedArray(loopMinType, uint.MaxValue)
        ));

        var loopMaxType = new IrArrayType(IrIntType.U32, 128);
        module.StaticVariables.Add(new IrStaticVariable(
            "__pgo_loop_max",
            loopMaxType,
            Visibility.Private,
            isMutable: true,
            initialValue: new IrArrayLiteral(loopMaxType)
        ));

        // Call site counters
        var callCounterType = new IrArrayType(IrIntType.U32, 512);
        module.StaticVariables.Add(new IrStaticVariable(
            "__pgo_call_counter",
            callCounterType,
            Visibility.Private,
            isMutable: true,
            initialValue: new IrArrayLiteral(callCounterType)
        ));
    }

    private void InstrumentFunctionEntry(IrFunction function, ushort funcId)
    {
        // Insert counter increment at function entry
        var firstBlock = function.BasicBlocks.FirstOrDefault();
        if (firstBlock == null) return;

        var counterInc = new IrProfileCounter(
            ProfileCounterType.FunctionEntry,
            funcId,
            $"Function: {function.Name}"
        );

        firstBlock.Instructions.Insert(0, counterInc);
    }

    private void InstrumentBranches(IrFunction function)
    {
        foreach (var block in function.BasicBlocks)
        {
            for (int i = 0; i < block.Instructions.Count; i++)
            {
                var instruction = block.Instructions[i];

                if (instruction is IrConditionalBranch condBranch)
                {
                    // Assign unique branch ID
                    int branchId = _nextBranchId++;
                    condBranch.BranchId = branchId;

                    // Record metadata
                    uint fileId = GetFileId(condBranch.Location?.FileName ?? "");
                    uint line = (uint)(condBranch.Location?.Line ?? 0);
                    _branchMetadata.Add(((ushort)branchId, fileId, line));

                    // We'll instrument the branch targets instead of the branch itself
                    // to avoid complex control flow modifications

                    // Create a new block for true path with counter
                    var trueBranchBlock = new IrBasicBlock($"{block.Label}_true_{branchId}");
                    trueBranchBlock.Instructions.Add(new IrProfileCounter(
                        ProfileCounterType.BranchTrue,
                        branchId,
                        $"Branch {branchId} true path"
                    ));
                    trueBranchBlock.Instructions.Add(new IrBranch(condBranch.TrueTarget));
                    function.BasicBlocks.Add(trueBranchBlock);

                    // Create a new block for false path with counter
                    var falseBranchBlock = new IrBasicBlock($"{block.Label}_false_{branchId}");
                    falseBranchBlock.Instructions.Add(new IrProfileCounter(
                        ProfileCounterType.BranchFalse,
                        branchId,
                        $"Branch {branchId} false path"
                    ));
                    falseBranchBlock.Instructions.Add(new IrBranch(condBranch.FalseTarget));
                    function.BasicBlocks.Add(falseBranchBlock);

                    // Update conditional branch targets
                    condBranch.TrueTarget = trueBranchBlock.Label;
                    condBranch.FalseTarget = falseBranchBlock.Label;
                }
            }
        }
    }

    private void InstrumentLoops(IrFunction function)
    {
        var cfg = new ControlFlowGraph(function);
        var loopDetector = new LoopDetector(cfg);
        var loops = loopDetector.DetectLoops();

        foreach (var loop in loops)
        {
            // Assign unique loop ID
            int loopId = _nextLoopId++;
            loop.LoopId = loopId;

            // Record metadata
            var headerBlock = loop.Header.Block;
            uint fileId = GetFileId(headerBlock?.Instructions.FirstOrDefault()?.Location?.FileName ?? "");
            uint line = (uint)(headerBlock?.Instructions.FirstOrDefault()?.Location?.Line ?? 0);
            _loopMetadata.Add(((ushort)loopId, fileId, line));

            // Instrument loop header (iteration counter)
            if (headerBlock != null)
            {
                var iterCounter = new IrProfileCounter(
                    ProfileCounterType.LoopIteration,
                    loopId,
                    $"Loop {loopId} iteration"
                );
                headerBlock.Instructions.Insert(0, iterCounter);
            }

            // Instrument loop exit (execution counter + min/max tracking)
            foreach (var exitEdge in loop.BackEdges)
            {
                // Find the block that exits the loop
                var exitBlock = exitEdge.Source.Block;
                if (exitBlock != null)
                {
                    var execCounter = new IrProfileCounter(
                        ProfileCounterType.LoopExecution,
                        loopId,
                        $"Loop {loopId} execution"
                    );
                    // Insert before the branch instruction
                    var lastInstrIndex = exitBlock.Instructions.Count - 1;
                    if (lastInstrIndex >= 0)
                    {
                        exitBlock.Instructions.Insert(lastInstrIndex, execCounter);
                    }
                }
            }
        }
    }

    private void InstrumentCallSites(IrFunction function)
    {
        foreach (var block in function.BasicBlocks)
        {
            for (int i = 0; i < block.Instructions.Count; i++)
            {
                var instruction = block.Instructions[i];

                if (instruction is IrCall call && !call.FunctionName.StartsWith("__pgo_"))
                {
                    // Assign unique call site ID
                    int callSiteId = _nextCallSiteId++;

                    // Record metadata
                    _callSiteMetadata.Add((
                        (ushort)callSiteId,
                        function.Name,
                        call.FunctionName
                    ));

                    // Insert counter before call
                    var counter = new IrProfileCounter(
                        ProfileCounterType.CallSite,
                        callSiteId,
                        $"Call to {call.FunctionName}"
                    );
                    block.Instructions.Insert(i, counter);
                    i++; // Skip the inserted counter
                }
            }
        }
    }

    private void AddProfileDumpAtExit(IrModule module)
    {
        // Create __pgo_dump_profile() function
        var dumpFunc = new IrFunction(
            "__pgo_dump_profile",
            IrVoidType.Instance,
            Visibility.Private,
            isExtern: true  // Implemented in libprofile.a
        );
        module.AddFunction(dumpFunc);

        // Call __pgo_dump_profile() at exit via atexit()
        // This is handled by the runtime library's constructor
    }

    private void SaveInstrumentationMetadata(IrModule module)
    {
        // Store metadata in module for use during profile data generation
        // This will be written to the .pgo file by the runtime

        var metadata = new InstrumentationMetadata
        {
            Functions = _functionMetadata,
            Branches = _branchMetadata,
            Loops = _loopMetadata,
            CallSites = _callSiteMetadata
        };

        // Serialize metadata to C arrays in generated code
        // (See C code generation section)
    }

    private uint GetFileId(string fileName)
    {
        // Simple hash of filename for file ID
        if (string.IsNullOrEmpty(fileName)) return 0;
        return (uint)fileName.GetHashCode() & 0x7FFFFFFF;
    }

    private IrArrayLiteral CreateInitializedArray(IrArrayType type, uint value)
    {
        var literal = new IrArrayLiteral(type);
        for (int i = 0; i < ((IrArrayType)type).Length; i++)
        {
            literal.Elements.Add(new IrConstant(value, IrIntType.U32));
        }
        return literal;
    }
}

{

    public List<(ushort Id, uint FileId, uint Line)> Loops { get; set; } = new();
    public List<(ushort Id, string Caller, string Callee)> CallSites { get; set; } = new();
```

```c
    u16 num_branches;
    u16 num_loops;
    u16 num_call_sites;
} __pgo_metadata_counts;

    u16 id;

    u16 id;

    const char *callee;

extern const char *__pgo_compiler_version;

#define PGO_MAGIC 0x4E50474F  /* "NPGO" */
#define PGO_VERSION 0x0001

/* Write profile data to novus_profile.pgo */
void __pgo_dump_profile(void) {
    BPTR file = Open("novus_profile.pgo", MODE_NEWFILE);
    if (!file) {
        /* Silently fail if can't write profile */
        return;
    }

    /* Write header */
    u32 magic = PGO_MAGIC;
    Write(file, &magic, 4);

    u16 version = PGO_VERSION;
    Write(file, &version, 2);

    /* Write source hash */
    Write(file, __pgo_source_hash, 32);

    /* Write timestamp (Unix epoch) */
    struct DateStamp ds;
    DateStamp(&ds);
    /* Convert Amiga timestamp (1978-01-01) to Unix (1970-01-01) */
    u64 timestamp = (u64)ds.ds_Days * 86400 +
                    (u64)ds.ds_Minute * 60 +
                    (u64)ds.ds_Tick / TICKS_PER_SECOND +
                    252460800;  /* Offset from Unix to Amiga epoch */
    Write(file, &timestamp, 8);

    /* Write compiler version */
    u8 version_len = strlen(__pgo_compiler_version);
    Write(file, &version_len, 1);
    Write(file, __pgo_compiler_version, version_len);

    /* Write function profiles */
    Write(file, &__pgo_metadata_counts.num_functions, 2);
    for (u16 i = 0; i < __pgo_metadata_counts.num_functions; i++) {
        Write(file, &__pgo_func_metadata[i].id, 2);
        u8 name_len = strlen(__pgo_func_metadata[i].name);
        Write(file, &name_len, 1);
        Write(file, __pgo_func_metadata[i].name, name_len);
        Write(file, &__pgo_func_counter[i], 4);

        /* TODO: Cycle counter (requires timer.device integration) */
        u32 cycles = 0;
        Write(file, &cycles, 4);
    }

    /* Write branch profiles */
    Write(file, &__pgo_metadata_counts.num_branches, 2);
    for (u16 i = 0; i < __pgo_metadata_counts.num_branches; i++) {
        Write(file, &__pgo_branch_metadata[i].id, 2);
        Write(file, &__pgo_branch_metadata[i].file_id, 4);
        Write(file, &__pgo_branch_metadata[i].line, 4);

        /* Branch true/false counts */
        u32 true_count = __pgo_branch_counter[i * 2];
        u32 false_count = __pgo_branch_counter[i * 2 + 1];
        Write(file, &true_count, 4);
        Write(file, &false_count, 4);
    }

    /* Write loop profiles */
    Write(file, &__pgo_metadata_counts.num_loops, 2);
    for (u16 i = 0; i < __pgo_metadata_counts.num_loops; i++) {
        Write(file, &__pgo_loop_metadata[i].id, 2);
        Write(file, &__pgo_loop_metadata[i].file_id, 4);
        Write(file, &__pgo_loop_metadata[i].line, 4);
        Write(file, &__pgo_loop_total[i], 4);
        Write(file, &__pgo_loop_executions[i], 4);
        Write(file, &__pgo_loop_min[i], 4);
        Write(file, &__pgo_loop_max[i], 4);
    }

    /* Write call site profiles */
    Write(file, &__pgo_metadata_counts.num_call_sites, 2);
    for (u16 i = 0; i < __pgo_metadata_counts.num_call_sites; i++) {
        Write(file, &__pgo_call_metadata[i].id, 2);
        u8 caller_len = strlen(__pgo_call_metadata[i].caller);
        Write(file, &caller_len, 1);
        Write(file, __pgo_call_metadata[i].caller, caller_len);
        u8 callee_len = strlen(__pgo_call_metadata[i].callee);
        Write(file, &callee_len, 1);
        Write(file, __pgo_call_metadata[i].callee, callee_len);
        Write(file, &__pgo_call_counter[i], 4);
    }

    Close(file);
}

/* Register cleanup handler */
__attribute__((constructor))
void __pgo_init(void) {
    atexit(__pgo_dump_profile);
}

static struct MsgPort *timer_port = NULL;
static struct timerequest *timer_req = NULL;

void __pgo_start_timer(void) {
    if (!timer_port) {
        timer_port = CreateMsgPort();
        if (timer_port) {
            timer_req = (struct timerequest *)CreateIORequest(timer_port, sizeof(struct timerequest));
            if (timer_req) {
                OpenDevice(TIMERNAME, UNIT_MICROHZ, (struct IORequest *)timer_req, 0);
            }
        }
    }

    if (!timer_req) return 0;

    struct EClockVal eclock;
    ReadEClock(&eclock);

    /* Convert E-clock ticks to cycles (68020 @ 14.32 MHz on A500+) */
    /* Adjust multiplier based on CPU speed */
    return eclock.ev_lo * 14;  /* Approximate */
}
```

### 5.1 Emitting Profile Counters

```csharp
// In CCodeGenerator.cs

private void EmitProfileCounter(IrProfileCounter counter)
{
    switch (counter.Type)
    {
        case ProfileCounterType.FunctionEntry:
            _output.AppendLine($"__pgo_func_counter[{counter.CounterId}]++;");
            break;

        case ProfileCounterType.BranchTrue:
        case ProfileCounterType.BranchFalse:
            int branchCounterIndex = counter.CounterId * 2 +
                (counter.Type == ProfileCounterType.BranchTrue ? 0 : 1);
            _output.AppendLine($"__pgo_branch_counter[{branchCounterIndex}]++;");
            break;

        case ProfileCounterType.LoopIteration:
            _output.AppendLine($"__pgo_loop_total[{counter.CounterId}]++;");
            break;

        case ProfileCounterType.LoopExecution:
            // Track min/max iterations
            _output.AppendLine($"{{");
            _output.AppendLine($"  u32 iters = __pgo_loop_total[{counter.CounterId}];");
            _output.AppendLine($"  if (iters < __pgo_loop_min[{counter.CounterId}])");
            _output.AppendLine($"    __pgo_loop_min[{counter.CounterId}] = iters;");
            _output.AppendLine($"  if (iters > __pgo_loop_max[{counter.CounterId}])");
            _output.AppendLine($"    __pgo_loop_max[{counter.CounterId}] = iters;");
            _output.AppendLine($"  __pgo_loop_executions[{counter.CounterId}]++;");
            _output.AppendLine($"}}");
            break;

        case ProfileCounterType.CallSite:
            _output.AppendLine($"__pgo_call_counter[{counter.CounterId}]++;");
            break;
    }
}
```

```csharp
private void EmitPgoMetadata(InstrumentationMetadata metadata)
{
    // Metadata counts
    _output.AppendLine("const struct {");
    _output.AppendLine("  u16 num_functions;");
    _output.AppendLine("  u16 num_branches;");

    _output.AppendLine("  const char *name;");

    _output.AppendLine();

    _output.AppendLine("  u32 line;");

    {
        _output.AppendLine($"  {{ {id}, {fileId}, {line} }},");
    }
    _output.AppendLine("};");
    _output.AppendLine();

    // Loop metadata
    _output.AppendLine("const struct {");
    _output.AppendLine("  u16 id;");
    _output.AppendLine("  u32 file_id;");
    _output.AppendLine("  u32 line;");
    _output.AppendLine($"}} __pgo_loop_metadata[{metadata.Loops.Count}] = {{");
    foreach (var (id, fileId, line) in metadata.Loops)
    {
        _output.AppendLine($"  {{ {id}, {fileId}, {line} }},");
    }
    _output.AppendLine("};");
    _output.AppendLine();

    // Call site metadata
    _output.AppendLine("const struct {");
    _output.AppendLine("  u16 id;");
    _output.AppendLine("  const char *caller;");
    _output.AppendLine("  const char *callee;");

    _output.AppendLine("};");
    _output.AppendLine();

    // Compiler version
    _output.AppendLine($"const char *__pgo_compiler_version = \"{_compilerVersion}\";");
}
```

This implementation guide provides the detailed code structures needed to implement PGO in the Novus compiler. The design balances:

- **Amiga RAM constraints** (efficient 8KB instrumentation footprint)
- **VBCC compatibility** (standard C99, no compiler-specific extensions)
- **Modularity** (each pass is independent and testable)
- **Performance** (minimal runtime overhead, ~5% for instrumented builds)
